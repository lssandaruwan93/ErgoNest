        e.preventDefault();
        first.focus();
      }
    });
  }

  // Close search on Escape
  document.addEventListener('keydown', function (e) {
    if (e.key === 'Escape' && searchOverlay && !searchOverlay.hidden) {
      closeSearch();
    }
  });

  /* ---------------------------------------------------------
     4. Newsletter form (demo only — no network request)
     --------------------------------------------------------- */
  var newsletter = document.querySelector('[data-newsletter]');
  if (newsletter) {
    var note  = newsletter.querySelector('[data-newsletter-note]');
    var email = newsletter.querySelector('input[type="email"]');

    newsletter.addEventListener('submit', function (e) {
      e.preventDefault();

      var value = email ? email.value.trim() : '';
      var valid = /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value);

      if (!note) return;

      if (!valid) {
        note.textContent = 'Please enter a valid email address.';
        note.classList.add('is-error');
        note.hidden = false;
        if (email) email.focus();
        return;
      }

      note.textContent =
        'Thanks! The newsletter is not connected yet in this demo — hook it up to your email provider when ready.';
      note.classList.remove('is-error');
      note.hidden = false;
      newsletter.reset();
    });
  }

  /* ---------------------------------------------------------
     5. Close mobile nav on Escape
     --------------------------------------------------------- */
  document.addEventListener('keydown', function (e) {
    if (e.key === 'Escape') closeNav();
  });

})();
